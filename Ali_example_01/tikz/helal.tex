% --- MACROS -------------------------------------------------------
\newcommand{\rotateAround}[7]{%
  \pgfmathsetmacro{\kkn}{sqrt((#4)*(#4)+(#5)*(#5)+(#6)*(#6))}%
  \pgfmathsetmacro{\kkx}{(#4)/\kkn}%
  \pgfmathsetmacro{\kky}{(#5)/\kkn}%
  \pgfmathsetmacro{\kkz}{(#6)/\kkn}%
  \pgfmathsetmacro{\co}{cos(#7)}\pgfmathsetmacro{\si}{sin(#7)}%
  \pgfmathsetmacro{\dt}{\kkx*(#1)+\kky*(#2)+\kkz*(#3)}%
  \pgfmathsetmacro{\cpx}{\kky*(#3)-\kkz*(#2)}%
  \pgfmathsetmacro{\cpy}{\kkz*(#1)-\kkx*(#3)}%
  \pgfmathsetmacro{\cpz}{\kkx*(#2)-\kky*(#1)}%
  \pgfmathsetmacro{\rRx}{(#1)*\co+\cpx*\si+\kkx*\dt*(1-\co)}%
  \pgfmathsetmacro{\rRy}{(#2)*\co+\cpy*\si+\kky*\dt*(1-\co)}%
  \pgfmathsetmacro{\rRz}{(#3)*\co+\cpz*\si+\kkz*\dt*(1-\co)}%
}

\newcommand{\drawCircle}[6]{%
  \begingroup
  \pgfmathsetmacro{\nrm}{sqrt((#1)*(#1)+(#2)*(#2)+(#3)*(#3))}%
  \pgfmathsetmacro{\nx}{(#1)/\nrm}\pgfmathsetmacro{\ny}{(#2)/\nrm}\pgfmathsetmacro{\nz}{(#3)/\nrm}%
  \pgfmathsetmacro{\nxy}{sqrt(\nx*\nx+\ny*\ny)}%
  \pgfmathparse{\nxy<0.001 ? 1 : 0}\let\nIsZ\pgfmathresult
  \ifnum\nIsZ=1
    \def\ux{1}\def\uy{0}\def\uz{0}\def\vx{0}\def\vy{1}\def\vz{0}%
  \else
    \pgfmathsetmacro{\ux}{ \ny/\nxy}\pgfmathsetmacro{\uy}{-\nx/\nxy}\def\uz{0}%
    \pgfmathsetmacro{\vx}{\ny*\uz-\nz*\uy}%
    \pgfmathsetmacro{\vy}{\nz*\ux-\nx*\uz}%
    \pgfmathsetmacro{\vz}{\nx*\uy-\ny*\ux}%
  \fi
  \drawSmallCircle{0}{0}{0}{#1}{#2}{#3}{#4}{#5}{#6}
  \endgroup
}

\newcommand{\drawSmallCircle}[9]{%
  \begingroup
  \pgfmathsetmacro{\nrm}{sqrt((#4)*(#4)+(#5)*(#5)+(#6)*(#6))}%
  \pgfmathsetmacro{\nx}{(#4)/\nrm}\pgfmathsetmacro{\ny}{(#5)/\nrm}\pgfmathsetmacro{\nz}{(#6)/\nrm}%
  \pgfmathsetmacro{\nxy}{sqrt(\nx*\nx+\ny*\ny)}%
  \pgfmathparse{\nxy<0.001 ? 1 : 0}\let\nIsZ\pgfmathresult
  \ifnum\nIsZ=1
    \def\ux{1}\def\uy{0}\def\uz{0}\def\vx{0}\def\vy{1}\def\vz{0}%
  \else
    \pgfmathsetmacro{\ux}{ \ny/\nxy}\pgfmathsetmacro{\uy}{-\nx/\nxy}\def\uz{0}%
    \pgfmathsetmacro{\vx}{\ny*\uz-\nz*\uy}%
    \pgfmathsetmacro{\vy}{\nz*\ux-\nx*\uz}%
    \pgfmathsetmacro{\vz}{\nx*\uy-\ny*\ux}%
  \fi
  % Camera direction D pointing to the eye in tikz-3dplot
  \pgfmathsetmacro{\dx}{sin(\mTheta)*sin(\mPhi)}%
  \pgfmathsetmacro{\dy}{-sin(\mTheta)*cos(\mPhi)}%
  \pgfmathsetmacro{\dz}{cos(\mTheta)}%
  % Dot products for visibility: (C + r(u cos t + v sin t)) . D > 0
  \pgfmathsetmacro{\Av}{\ux*\dx+\uy*\dy+\uz*\dz}%
  \pgfmathsetmacro{\Bv}{\vx*\dx+\vy*\dy+\vz*\dz}%
  \pgfmathsetmacro{\CvD}{(#1)*\dx+(#2)*\dy+(#3)*\dz}%
  \pgfmathsetmacro{\Lv}{sqrt(\Av*\Av + \Bv*\Bv + 0.000001)}%
  \pgfmathsetmacro{\Argv}{min(1, max(-1, -\CvD / (#7 * \Lv)))}%
  \pgfmathsetmacro{\alphav}{atan2(\Bv, \Av)}%
  \pgfmathsetmacro{\dalpha}{acos(\Argv)}%
  % Visibility range [alpha - dalpha, alpha + dalpha]
  \pgfmathsetmacro{\tA}{\alphav - \dalpha}%
  \pgfmathsetmacro{\tB}{\alphav + \dalpha}%
  \pgfmathsetmacro{\tC}{\tA + 360}%
  % Check if circle is entirely visible or hidden
  \pgfmathparse{-\CvD / (#7 * \Lv) < -1 ? 1 : 0}\let\allVis\pgfmathresult
  \pgfmathparse{-\CvD / (#7 * \Lv) > 1 ? 1 : 0}\let\allHid\pgfmathresult
  \ifnum\allVis=1
    \draw[#8] plot[domain=0:360,samples=80,smooth,variable=\t]
      ({(#1)+(#7)*(\ux*cos(\t)+\vx*sin(\t))},{(#2)+(#7)*(\uy*cos(\t)+\vy*sin(\t))},{(#3)+(#7)*(\uz*cos(\t)+\vz*sin(\t))});
  \else\ifnum\allHid=1
    \begin{pgfonlayer}{back}
      \draw[#9] plot[domain=0:360,samples=80,smooth,variable=\t]
        ({(#1)+(#7)*(\ux*cos(\t)+\vx*sin(\t))},{(#2)+(#7)*(\uy*cos(\t)+\vy*sin(\t))},{(#3)+(#7)*(\uz*cos(\t)+\vz*sin(\t))});
    \end{pgfonlayer}
  \else
    \draw[#8] plot[domain=\tA:\tB,samples=80,smooth,variable=\t]
      ({(#1)+(#7)*(\ux*cos(\t)+\vx*sin(\t))},{(#2)+(#7)*(\uy*cos(\t)+\vy*sin(\t))},{(#3)+(#7)*(\uz*cos(\t)+\vz*sin(\t))});
    \begin{pgfonlayer}{back}
      \draw[#9] plot[domain=\tB:\tC,samples=80,smooth,variable=\t]
        ({(#1)+(#7)*(\ux*cos(\t)+\vx*sin(\t))},{(#2)+(#7)*(\uy*cos(\t)+\vy*sin(\t))},{(#3)+(#7)*(\uz*cos(\t)+\vz*sin(\t))});
    \end{pgfonlayer}
  \fi\fi
  \endgroup
}
